cmake_minimum_required(VERSION 3.16)
project(bindu_bimanual_qp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(pinocchio REQUIRED)
find_package(proxsuite REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(${PROJECT_NAME} SHARED src/core.cpp src/c_api.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE include)
target_link_libraries(${PROJECT_NAME} PRIVATE pinocchio::pinocchio proxsuite::proxsuite Eigen3::Eigen)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_bimanual_core test/test_core.cpp)
  target_include_directories(test_bimanual_core PRIVATE include)
  target_link_libraries(test_bimanual_core ${PROJECT_NAME} pinocchio::pinocchio proxsuite::proxsuite Eigen3::Eigen)
  target_compile_definitions(test_bimanual_core PRIVATE TEST_URDF_PATH="${CMAKE_CURRENT_SOURCE_DIR}/test/two_arms.urdf")
endif()
ament_package()
