cmake_minimum_required(VERSION 3.16)
project(bindu_navigation LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(bindu_interfaces REQUIRED)
add_executable(nav2_velocity_identity native/nav2_velocity_identity.cpp)
target_compile_features(nav2_velocity_identity PRIVATE cxx_std_17)
ament_target_dependencies(nav2_velocity_identity rclcpp geometry_msgs std_msgs bindu_interfaces)
install(TARGETS nav2_velocity_identity DESTINATION lib/${PROJECT_NAME})
ament_package()
